local function GetThemeColor()
    local t = tick() % 8
    if t < 2 then return Color3.fromRGB(100, 100, 255)
    elseif t < 4 then t = t - 2; return Color3.fromRGB(100, 200, 255)
    elseif t < 6 then t = t - 4; return Color3.fromRGB(200, 100, 255)
    else t = t - 6; return Color3.fromRGB(255, 100, 200) end
end

    page.CanvasSize = UDim2.new(0, 0, 0, 500)
    page.Parent = ContentArea
    ContentPages[name] = page
    
    local pageLayout = Instance.new("UIListLayout", page)
    pageLayout.Padding = UDim.new(0, 8)
    pageLayout.HorizontalAlignment = Enum.HorizontalAlignment.Center
    pageLayout.SortOrder = Enum.SortOrder.LayoutOrder
    
    btn.MouseButton1Click:Connect(function()
        for _, b in pairs(SidebarButtons) do
            b.BackgroundColor3 = Color3.fromRGB(20, 20, 26)
            b.TextColor3 = RF.SubText
        end
        btn.BackgroundColor3 = RF.Accent
        btn.BackgroundTransparency = 0.2
        btn.TextColor3 = RF.Text
        for _, p in pairs(ContentPages) do p.Visible = false end
        page.Visible = true
        State.CurrentTab = name
    end)
    
    SidebarButtons[name] = btn
    return page, btn
end

    switchKnob.Size = UDim2.new(0, 18, 0, 18)
    switchKnob.Position = UDim2.new(0, 2, 0.5, -9)
    switchKnob.BackgroundColor3 = RF.Text
    switchKnob.BorderSizePixel = 0
    Instance.new("UICorner", switchKnob).CornerRadius = UDim.new(1, 0)
    
    local switchBtn = Instance.new("TextButton", toggle)
    switchBtn.Size = UDim2.new(0, 44, 0, 22)
    switchBtn.Position = UDim2.new(1, -56, 0.5, -11)
    switchBtn.BackgroundTransparency = 1
    switchBtn.Text = ""
    switchBtn.BorderSizePixel = 0
    switchBtn.ZIndex = 5
    
    local function update()
        local state = getState()
        switchBg.BackgroundColor3 = state and RF.ToggleOn or RF.ToggleOff
        switchKnob.Position = state and UDim2.new(1, -20, 0.5, -9) or UDim2.new(0, 2, 0.5, -9)
    end
    
    switchBtn.MouseButton1Click:Connect(function()
        setState(not getState())
        update()
    end)
    
    task.spawn(function()
        while task.wait(0.2) do update() end
    end)
    
    return toggle
end

local function CreateSlider(parent, text, min, max, getValue, setValue)
    local slider = Instance.new("Frame")
    slider.Size = UDim2.new(1, 0, 0, 60)
    slider.BackgroundColor3 = RF.Section
    slider.BorderSizePixel = 0
    slider.Parent = parent
    Instance.new("UICorner", slider).CornerRadius = UDim.new(0, 6)
    
    local label = Instance.new("TextLabel", slider)
    label.Size = UDim2.new(0.5, 0, 0, 20)
    label.Position = UDim2.new(0, 12, 0, 6)
    label.BackgroundTransparency = 1
    label.Text = text
    label.TextColor3 = RF.Text
    label.TextSize = 12
    label.Font = Enum.Font.GothamMedium
    label.TextXAlignment = Enum.TextXAlignment.Left
    
    local valLabel = Instance.new("TextLabel", slider)
    valLabel.Size = UDim2.new(0, 50, 0, 20)
    valLabel.Position = UDim2.new(1, -62, 0, 6)
    valLabel.BackgroundTransparency = 1
    valLabel.Text = tostring(getValue())
    valLabel.TextColor3 = RF.Accent
    valLabel.TextSize = 12
    valLabel.Font = Enum.Font.GothamBold
    valLabel.TextXAlignment = Enum.TextXAlignment.Right
    
    local bg = Instance.new("Frame", slider)
    bg.Size = UDim2.new(1, -24, 0, 6)
    bg.Position = UDim2.new(0, 12, 0, 34)
    bg.BackgroundColor3 = Color3.fromRGB(40, 40, 48)
    bg.BorderSizePixel = 0
    Instance.new("UICorner", bg).CornerRadius = UDim.new(0, 3)
    
    local fill = Instance.new("Frame", bg)
    fill.Size = UDim2.new((getValue() - min) / (max - min), 0, 1, 0)
    fill.BackgroundColor3 = RF.Slider
    fill.BorderSizePixel = 0
    Instance.new("UICorner", fill).CornerRadius = UDim.new(0, 3)
    
    local knob = Instance.new("Frame", bg)
    knob.Size = UDim2.new(0, 16, 0, 16)
    knob.Position = UDim2.new((getValue() - min) / (max - min), -8, 0, -5)
    knob.BackgroundColor3 = RF.Text
    knob.BorderSizePixel = 0
    knob.ZIndex = 10
    Instance.new("UICorner", knob).CornerRadius = UDim.new(1, 0)
    
    local dragging = false
    local function update(x)
        local r = math.clamp((x - bg.AbsolutePosition.X) / bg.AbsoluteSize.X, 0, 1)
        fill.Size = UDim2.new(r, 0, 1, 0)
        knob.Position = UDim2.new(r, -8, 0, -5)
        local v = math.floor(r * (max - min) + min)
        setValue(v)
        valLabel.Text = tostring(v)
    end
    
    bg.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true; update(i.Position.X)
        end
    end)
    knob.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true
        end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            update(i.Position.X)
        end
    end)
    UserInputService.InputEnded:Connect(function() dragging = false end)
end

function StartFly()
    StopFly()
    local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
    local root = char:WaitForChild("HumanoidRootPart")
    local hum = char:WaitForChild("Humanoid")
    if not root or not hum then return end
    
    State.FlyGyro = Instance.new("BodyGyro")
    State.FlyGyro.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    State.FlyGyro.D = 50; State.FlyGyro.P = 3000
    State.FlyGyro.CFrame = root.CFrame
    State.FlyGyro.Parent = root
    
    State.FlyVel = Instance.new("BodyVelocity")
    State.FlyVel.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    State.FlyVel.Velocity = Vector3.zero
    State.FlyVel.P = 5000
    State.FlyVel.Parent = root
    
    State.FlyBodyPos = Instance.new("BodyPosition")
    State.FlyBodyPos.MaxForce = Vector3.new(0, 9e9, 0)
    State.FlyBodyPos.P = 50000; State.FlyBodyPos.D = 1000
    State.FlyBodyPos.Position = root.Position
    State.FlyBodyPos.Parent = root
    
    hum.PlatformStand = true
    local lastHeight = root.Position.Y
    
    State.FlyConn = RunService.Heartbeat:Connect(function()
        if not State.Fly then StopFly(); return end
        local c = LocalPlayer.Character
        if not c then StopFly(); return end
        local r = c:FindFirstChild("HumanoidRootPart")
        if not r then StopFly(); return end
        if not State.FlyGyro or not State.FlyGyro.Parent then StopFly(); return end
        if not State.FlyVel or not State.FlyVel.Parent then StopFly(); return end
        if not State.FlyBodyPos or not State.FlyBodyPos.Parent then StopFly(); return end
        
        State.FlyGyro.CFrame = Camera.CFrame
        local dir = Vector3.zero
        local verticalInput = false
        
        if isMobile then
            if State.JoyDir.Magnitude > 0.1 then
                dir = (Camera.CFrame.LookVector * -State.JoyDir.Z) + (Camera.CFrame.RightVector * State.JoyDir.X)
            end
            if State.VertIn ~= 0 then
                verticalInput = true
                lastHeight = r.Position.Y + State.VertIn * State.FlySpeed * 0.05
            end
        else
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir += Camera.CFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir -= Camera.CFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir -= Camera.CFrame.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir += Camera.CFrame.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then verticalInput = true; lastHeight = r.Position.Y + State.FlySpeed * 0.05 end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then verticalInput = true; lastHeight = r.Position.Y - State.FlySpeed * 0.05 end
        end
        
        State.FlyVel.Velocity = dir.Magnitude > 0 and dir.Unit * State.FlySpeed or Vector3.zero
        if verticalInput or dir.Magnitude < 0.1 then
            State.FlyBodyPos.Position = Vector3.new(r.Position.X, lastHeight, r.Position.Z)
            if dir.Magnitude < 0.1 then State.FlyVel.Velocity = Vector3.zero end
        else
            State.FlyBodyPos.Position = Vector3.new(r.Position.X, r.Position.Y, r.Position.Z)
            lastHeight = r.Position.Y
        end
    end)
end

function StopFly()
    if State.FlyConn then State.FlyConn:Disconnect(); State.FlyConn = nil end
    if State.FlyGyro then State.FlyGyro:Destroy(); State.FlyGyro = nil end
    if State.FlyVel then State.FlyVel:Destroy(); State.FlyVel = nil end
    if State.FlyBodyPos then State.FlyBodyPos:Destroy(); State.FlyBodyPos = nil end
    pcall(function() local c = LocalPlayer.Character; if c and c:FindFirstChild("Humanoid") then c.Humanoid.PlatformStand = false end end)
end

function StartNoclip()
    StopNoclip()
    State.NoclipConn = RunService.Heartbeat:Connect(function()
        if not State.Noclip then StopNoclip(); return end
        local c = LocalPlayer.Character
        if c then for _, p in ipairs(c:GetDescendants()) do if p:IsA("BasePart") and p.CanCollide then p.CanCollide = false end end end
    end)
end
function StopNoclip() if State.NoclipConn then State.NoclipConn:Disconnect(); State.NoclipConn = nil end end

end

function StartInfiniteJump()
    StopInfiniteJump()

    end)
end
function StopInfiniteJump() if State.JumpConn then State.JumpConn:Disconnect(); State.JumpConn = nil end end

function StartAntiAFK()
    StopAntiAFK()
    State.AntiAFKConn = RunService.Heartbeat:Connect(function()
        if not State.AntiAFK then StopAntiAFK(); return end
        VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.LeftControl, false, game)
        task.wait(0.1)
        VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.LeftControl, false, game)
    end)
end
function StopAntiAFK() if State.AntiAFKConn then State.AntiAFKConn:Disconnect(); State.AntiAFKConn = nil end end

function StartAimbot()
    StopAimbot()
    State.AimbotConn = RunService.Heartbeat:Connect(function()
        if not State.Aimbot then StopAimbot(); return end
        if not State.AimbotTarget or not State.AimbotTarget.Character or not State.AimbotTarget.Character:FindFirstChild("Head") then return end
        local targetHead = State.AimbotTarget.Character.Head
        local camPos = Camera.CFrame.Position
        local targetPos = targetHead.Position
        local screenPos, onScreen = Camera:WorldToViewportPoint(targetPos)
        if not onScreen then return end
        local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
        if (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude > State.AimbotFOV then return end
        Camera.CFrame = Camera.CFrame:Lerp(CFrame.new(camPos, targetPos), State.AimbotSmoothness / 100)
    end)
end
function StopAimbot() if State.AimbotConn then State.AimbotConn:Disconnect(); State.AimbotConn = nil end end

function UpdateESP()
    for _, v in ipairs(State.ESPFolder:GetChildren()) do v:Destroy() end
    if not State.ESP then return end
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local h = Instance.new("Highlight")
            h.Name = p.Name; h.Adornee = p.Character
            h.FillColor = ESPColors[State.ESPColor][1]; h.FillTransparency = 0.4
            h.OutlineColor = Color3.fromRGB(255, 255, 255); h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            h.Parent = State.ESPFolder
        end
    end
end

function SetInvis(s)
    local c = LocalPlayer.Character; if not c then return end
    for _, d in ipairs(c:GetDescendants()) do
        if d:IsA("BasePart") then d.Transparency = s and 1 or 0
        elseif d:IsA("Decal") or d:IsA("Texture") then d.Transparency = s and 1 or 0 end
    end
end

function CreateMobileFly()
    RemoveMobileFly()
    local M = Instance.new("ScreenGui"); M.Name = "MobileV13"; M.Parent = CoreGui; M.ResetOnSpawn = false; M.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    State.Mobile = M; _G.MobileV13 = M
    
    local tz = Instance.new("Frame", M); tz.Size = UDim2.new(0, 300, 1, -80); tz.Position = UDim2.new(0, 0, 0, 40)
    tz.BackgroundTransparency = 1; tz.Active = true; tz.ZIndex = 50000
    
    local jv = Instance.new("Frame", M); jv.Size = UDim2.new(0, 160, 0, 160); jv.Position = UDim2.new(0, 25, 0.82, -180)
    jv.BackgroundColor3 = Color3.fromRGB(18, 18, 24); jv.BackgroundTransparency = 0.5
    jv.Visible = false; jv.ZIndex = 50001
    Instance.new("UICorner", jv).CornerRadius = UDim.new(1, 0)
    local jvs = Instance.new("UIStroke", jv); jvs.Color = RF.Accent; jvs.Thickness = 2
    
    local jt = Instance.new("Frame", jv); jt.Size = UDim2.new(0, 60, 0, 60); jt.Position = UDim2.new(0.5, -30, 0.5, -30)
    jt.BackgroundColor3 = RF.Accent; jt.BackgroundTransparency = 0.4; jt.ZIndex = 50002
    Instance.new("UICorner", jt).CornerRadius = UDim.new(1, 0)
    
    local drag, center = false, nil
    tz.InputBegan:Connect(function(i) if i.UserInputType == Enum.UserInputType.Touch then drag = true; center = i.Position; jv.Visible = true; jv.Position = UDim2.new(0, center.X - 80, 0, center.Y - 80) end end)
    UserInputService.InputChanged:Connect(function(i)
        if not drag or i.UserInputType ~= Enum.UserInputType.Touch then return end
        local d = i.Position - center; local m = 50; if d.Magnitude > m then d = d.Unit * m end
        jt.Position = UDim2.new(0, d.X + 80 - 30, 0, d.Y + 80 - 30)
        State.JoyDir = Vector3.new(d.X/m, 0, -d.Y/m)
    end)
    UserInputService.InputEnded:Connect(function(i) if i.UserInputType == Enum.UserInputType.Touch then drag = false; jv.Visible = false; jt.Position = UDim2.new(0.5, -30, 0.5, -30); State.JoyDir = Vector3.zero end end)
    
    for _, data in ipairs({{"▲", 1, -370}, {"▼", -1, -220}}) do
        local btn = Instance.new("TextButton", M); btn.Size = UDim2.new(0, 70, 0, 70); btn.Position = UDim2.new(1, -90, 1, data[3])
        btn.BackgroundColor3 = RF.Background; btn.BackgroundTransparency = 0.4
        btn.Text = data[1]; btn.TextColor3 = RF.Accent; btn.TextSize = 30; btn.Font = Enum.Font.GothamBold; btn.AutoButtonColor = false; btn.ZIndex = 50000
        Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 14); Instance.new("UIStroke", btn).Color = RF.Accent; Instance.new("UIStroke", btn).Thickness = 1.5
        btn.InputBegan:Connect(function(i) if i.UserInputType == Enum.UserInputType.Touch then State.VertIn = data[2] end end)
        btn.InputEnded:Connect(function() State.VertIn = 0 end)
    end
end

function RemoveMobileFly()
    if State.Mobile then State.Mobile:Destroy(); State.Mobile = nil end
    State.JoyDir = Vector3.zero; State.VertIn = 0
end

    if State.Fly then State.Fly = false; State.Fly = true; StartFly() end
    if State.Noclip then State.Noclip = false; State.Noclip = true; StartNoclip() end
    if State.God then State.God = false; State.God = true; StartGodMode() end
    if State.SpeedHack then State.SpeedHack = false; State.SpeedHack = true; StartSpeedHack() end
    if State.JumpPower then ApplyJumpPower() end
    if State.InfiniteJump then State.InfiniteJump = false; State.InfiniteJump = true; StartInfiniteJump() end
    if State.Invis then SetInvis(true) end
    if State.ESP then UpdateESP() end
    if State.Aimbot then State.Aimbot = false; State.Aimbot = true; StartAimbot() end
end)
